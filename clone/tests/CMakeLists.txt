cmake_minimum_required(VERSION 3.20)
project(ocg_clone_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(OCG_CLONE_PROFILING "Compile clone phase timers and counters" OFF)
option(OCG_CLONE_UBSAN "Run clone integration with undefined behavior sanitizer" OFF)
option(OCG_CLONE_TSAN "Run clone tests with thread sanitizer" OFF)
set(CARDSCRIPTS_DIR "" CACHE PATH "Pinned CardScripts checkout")
get_filename_component(CORE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(LUA "${CORE}/lua/src")
include("${CORE}/clone/sources.cmake")
set(CORE_SOURCES card duel effect field interpreter libcard libdebug libduel
    libeffect libgroup ocgapi operations playerop processor processor_visit scriptlib)
set(LUA_SOURCES lapi lauxlib lbaselib lcode lctype ldebug ldo ldump lfunc lgc
    liolib llex lmathlib lmem lobject lopcodes lparser lstate lstring lstrlib
    ltable ltablib ltm lundump lvm lzio)
set(SOURCES ${OCG_CLONE_SOURCES})
foreach(NAME IN LISTS CORE_SOURCES)
    list(APPEND SOURCES "${CORE}/${NAME}.cpp")
endforeach()
foreach(NAME IN LISTS LUA_SOURCES)
    list(APPEND SOURCES "${LUA}/${NAME}.c")
    set_source_files_properties("${LUA}/${NAME}.c" PROPERTIES LANGUAGE CXX)
    if(MSVC)
        set_property(SOURCE "${LUA}/${NAME}.c" APPEND PROPERTY COMPILE_OPTIONS "/FI${CORE}/lua/luaconf-customize.h")
    else()
        set_property(SOURCE "${LUA}/${NAME}.c" APPEND PROPERTY COMPILE_OPTIONS "-include${CORE}/lua/luaconf-customize.h")
    endif()
endforeach()
add_library(clone_core STATIC ${SOURCES})
target_link_libraries(clone_core PUBLIC ocg_clone_boost ${CMAKE_DL_LIBS})
target_include_directories(clone_core PUBLIC "${CORE}" "${LUA}")
target_compile_definitions(clone_core PUBLIC PROCESSOR_VARIANT_CAP=20)
if(OCG_CLONE_PROFILING)
    target_compile_definitions(clone_core PUBLIC OCG_CLONE_PROFILING=1)
endif()
add_executable(clone_profiling profiling.cpp)
add_executable(clone_integration integration.cpp)
add_executable(clone_chain_reset chain_reset.cpp)
add_executable(clone_checkpoint checkpoint.cpp)
add_executable(clone_persistence persistence.cpp)
add_executable(clone_threads threads.cpp)
add_executable(clone_group_cursor group_cursor.cpp)
target_link_libraries(clone_persistence PRIVATE clone_core)
target_link_libraries(clone_group_cursor PRIVATE clone_core)
target_link_libraries(clone_checkpoint PRIVATE clone_core Threads::Threads)
find_package(Threads REQUIRED)
target_link_libraries(clone_profiling PRIVATE clone_core Threads::Threads)
target_link_libraries(clone_integration PRIVATE clone_core Threads::Threads)
target_link_libraries(clone_threads PRIVATE clone_core Threads::Threads)
target_link_libraries(clone_chain_reset PRIVATE clone_core)
if(UNIX)
    target_link_libraries(clone_integration PRIVATE m)
endif()
if(OCG_CLONE_UBSAN)
    if(MSVC)
        message(FATAL_ERROR "UBSan mode requires GCC or Clang")
    endif()
    foreach(TARGET clone_core clone_integration clone_chain_reset clone_profiling clone_checkpoint clone_persistence clone_threads clone_group_cursor)
        target_compile_options(${TARGET} PUBLIC -fsanitize=undefined -fno-sanitize-recover=all)
        target_link_options(${TARGET} PUBLIC -fsanitize=undefined -fno-sanitize-recover=all)
    endforeach()
endif()
if(OCG_CLONE_TSAN)
    if(MSVC)
        message(FATAL_ERROR "TSan mode requires GCC or Clang")
    endif()
    foreach(TARGET clone_core clone_integration clone_chain_reset clone_profiling clone_checkpoint clone_persistence clone_threads clone_group_cursor)
        target_compile_options(${TARGET} PUBLIC -fsanitize=thread -g)
        target_link_options(${TARGET} PUBLIC -fsanitize=thread)
    endforeach()
endif()
enable_testing()
add_test(NAME clone_profiling COMMAND clone_profiling)
add_test(NAME clone_chain_reset COMMAND clone_chain_reset)
if(CARDSCRIPTS_DIR)
    add_test(NAME clone_persistence COMMAND clone_persistence "${CARDSCRIPTS_DIR}" "${CMAKE_CURRENT_LIST_DIR}/fixtures")
    add_test(NAME clone_checkpoint COMMAND clone_checkpoint "${CARDSCRIPTS_DIR}" "${CMAKE_CURRENT_LIST_DIR}/fixtures")
    add_test(NAME clone_integration COMMAND clone_integration "${CARDSCRIPTS_DIR}" "${CMAKE_CURRENT_LIST_DIR}/fixtures")
    add_test(NAME clone_threads COMMAND clone_threads "${CARDSCRIPTS_DIR}" "${CMAKE_CURRENT_LIST_DIR}/fixtures")
    add_test(NAME clone_group_cursor COMMAND clone_group_cursor "${CARDSCRIPTS_DIR}" "${CMAKE_CURRENT_LIST_DIR}/fixtures")
endif()
